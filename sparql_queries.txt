PREFIX conf: <http://data-semantics-2526.org/acled/ontology#>
PREFIX res: <http://data-semantics-2526.org/acled/resource/>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

# =====================================================================
# ESEMPIO 1: Estrazione base di tutti gli eventi
# =====================================================================
# Recupera l'URI dell'evento, la data, la località e il tipo di evento.
SELECT ?event ?date ?location ?eventType
WHERE {
  ?event a conf:ConflictEvent ;
         conf:eventDate ?date ;
         conf:locationName ?location ;
         conf:eventType ?eventType .
}
ORDER BY DESC(?date)
LIMIT 100

# =====================================================================
# ESEMPIO 2: Filtro tramite stringhe e regex
# =====================================================================
# Trova tutti gli eventi avvenuti ad "Abu Dhabi" che menzionano
# la parola "missile" nelle note.
SELECT ?event ?date ?notes
WHERE {
  ?event a conf:ConflictEvent ;
         conf:locationName "Abu Dhabi"^^xsd:string ;
         conf:eventDate ?date ;
         conf:notes ?notes .
         
  FILTER(REGEX(?notes, "missile", "i"))
}
ORDER BY ?date

# =====================================================================
# ESEMPIO 3: Eventi che coinvolgono un attore specifico
# =====================================================================
# Cerca tutti gli eventi in cui le Forze Militari degli UAE
# sono state coinvolte (sia come Actor 1 che come Actor 2).
SELECT ?event ?date ?subEventType
WHERE {
  ?event a conf:ConflictEvent ;
         conf:eventDate ?date ;
         conf:subEventType ?subEventType .
         
  { ?event conf:hasActor1 <http://data-semantics-2526.org/acled/resource/actor/military-forces-of-the-united-arab-emirates-2004-2022> . }
  UNION
  { ?event conf:hasActor2 <http://data-semantics-2526.org/acled/resource/actor/military-forces-of-the-united-arab-emirates-2004-2022> . }
}

# =====================================================================
# ESEMPIO 4: Statistiche e Aggregazione (GROUP BY)
# =====================================================================
# Conta quanti eventi sono stati registrati per ogni "subEventType"
# e ordinali in modo decrescente.
SELECT ?subEventType (COUNT(?event) AS ?eventCount)
WHERE {
  ?event a conf:ConflictEvent ;
         conf:subEventType ?subEventType .
}
GROUP BY ?subEventType
ORDER BY DESC(?eventCount)

# =====================================================================
# ESEMPIO 5: Interrogazioni con range temporale
# =====================================================================
# Estrae eventi occorsi tra il 1 Gennaio 2022 e il 31 Gennaio 2022, 
# recuperando gli URI degli attori coinvolti, omettendoli se assenti (OPTIONAL).
SELECT ?event ?date ?actor1 ?actor2
WHERE {
  ?event a conf:ConflictEvent ;
         conf:eventDate ?date .
         
  OPTIONAL { ?event conf:hasActor1 ?actor1 . }
  OPTIONAL { ?event conf:hasActor2 ?actor2 . }
  
  FILTER(?date >= "2022-01-01"^^xsd:date && ?date <= "2022-01-31"^^xsd:date)
}
ORDER BY ?date
